<script>

let messages = [];

function render(){
    const box = document.getElementById("messages");
    box.innerHTML = "";

    messages.forEach((m,i)=>{
        box.innerHTML += `
        <div class="msg ${m.me ? 'me':'other'}">
            ${m.text}
            <div class="options" onclick="toggleMenu(${i})">⋮</div>

            <div class="menu" id="menu-${i}">
                <div onclick="editMsg(${i})">Edit</div>
                <div onclick="deleteMe(${i})">Delete for me</div>
                <div onclick="deleteAll(${i})">Delete for everyone</div>
            </div>
        </div>
        `;
    });

    box.scrollTop = box.scrollHeight;
}

function sendMsg(){
    const input = document.getElementById("msgInput");

    if(input.value.trim() === "") return;

    messages.push({text: input.value, me:true});
    input.value="";
    render();
}

function toggleMenu(i){
    const menu = document.getElementById(`menu-${i}`);
    menu.style.display = menu.style.display === "block" ? "none":"block";
}

function editMsg(i){
    let newText = prompt("Edit message:");
    if(newText){
        messages[i].text = newText;
        render();
    }
}

function deleteMe(i){
    messages.splice(i,1);
    render();
}

function deleteAll(i){
    messages[i].text = "Message deleted";
    render();
}

document.getElementById("msgInput").addEventListener("keypress", function(e){
    if(e.key === "Enter"){
        sendMsg();
    }
});

function goBack(){
    window.history.back();
}

function toggleSidebar(){
    const sidebar = document.getElementById("sidebar");
    sidebar.classList.toggle("closed");
}

</script>
